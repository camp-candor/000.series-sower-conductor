import { ContinuityCheckResult } from '../fce/lifecycle.interface.js';

export async function verifyLocalContinuity(
  sceneId: string,
  sqliteDb: any
): Promise<ContinuityCheckResult> {
  const startTime = Date.now();

  if (!sqliteDb || typeof sqliteDb.prepare !== 'function') {
    // In-memory mock fallback when no SQLite instance is provided
    return {
      clean: true,
      sceneId,
      dirtyCount: 0,
      totalDependencies: 0,
      latencyMs: Date.now() - startTime,
    };
  }

  const query = `
    SELECT 
      s.id AS scene_id,
      s.is_dirty AS scene_dirty,
      COUNT(sfd.fact_id) AS total_dependencies,
      SUM(CASE WHEN f.version > sfd.pinned_version THEN 1 ELSE 0 END) AS dirty_dependencies
    FROM scenes s
    LEFT JOIN scene_fact_dependencies sfd ON s.id = sfd.scene_id
    LEFT JOIN facts f ON sfd.fact_id = f.id
    WHERE s.id = ?
    GROUP BY s.id;
  `;

  try {
    const stmt = sqliteDb.prepare(query);
    const row = stmt.get(sceneId);

    if (!row) {
      return {
        clean: false,
        sceneId,
        dirtyCount: 0,
        totalDependencies: 0,
        latencyMs: Date.now() - startTime,
        reason: `SCENE_NOT_FOUND: Scene ${sceneId} not registered in local ledger`,
      };
    }

    const dirtyCount = Number(row.dirty_dependencies) || 0;
    const isSceneDirty = Number(row.scene_dirty) === 1;
    const isClean = dirtyCount === 0 && !isSceneDirty;

    return {
      clean: isClean,
      sceneId,
      dirtyCount: dirtyCount + (isSceneDirty ? 1 : 0),
      totalDependencies: Number(row.total_dependencies) || 0,
      latencyMs: Date.now() - startTime,
      reason: isClean ? undefined : 'DIRTY_SCENE_REJECTED: Unratified fact deltas exist',
    };
  } catch (err: any) {
    return {
      clean: false,
      sceneId,
      dirtyCount: 1,
      totalDependencies: 0,
      latencyMs: Date.now() - startTime,
      reason: `QUERY_ERROR: ${err.message}`,
    };
  }
}

import crypto from 'node:crypto';
import { CanonLockResult } from '../fce/lifecycle.interface.js';

export function deepFreeze<T>(obj: T): Readonly<T> {
  if (obj === null || typeof obj !== 'object') {
    return obj;
  }

  const propNames = Object.getOwnPropertyNames(obj);
  for (const name of propNames) {
    const value = (obj as any)[name];
    if (value && typeof value === 'object') {
      deepFreeze(value);
    }
  }

  return Object.freeze(obj);
}

export function canonicalJsonStringify(obj: unknown): string {
  if (obj === null || typeof obj !== 'object') {
    return JSON.stringify(obj);
  }

  if (Array.isArray(obj)) {
    return '[' + obj.map((item) => canonicalJsonStringify(item)).join(',') + ']';
  }

  const sortedKeys = Object.keys(obj as Record<string, unknown>).sort();
  const entries: string[] = [];

  for (const key of sortedKeys) {
    const val = (obj as Record<string, unknown>)[key];
    if (val !== undefined) {
      entries.push(JSON.stringify(key) + ':' + canonicalJsonStringify(val));
    }
  }

  return '{' + entries.join(',') + '}';
}

export function verifyCanonMerkleLock(
  canonState: unknown,
  expectedMerkleRoot: string
): CanonLockResult {
  // 1. Recursive Immutability Lock
  deepFreeze(canonState);

  // 2. Deterministic Hash Minting
  const canonicalString = canonicalJsonStringify(canonState);
  const calculatedMerkleRoot = crypto
    .createHash('sha256')
    .update(canonicalString)
    .digest('hex');

  const valid = calculatedMerkleRoot.toLowerCase() === expectedMerkleRoot.toLowerCase();

  return {
    valid,
    calculatedMerkleRoot,
    expectedMerkleRoot,
    diverged: !valid,
  };
}

export const runInceptionPreflight = async (
  cpy: LifecycleModel,
  bal: LifecycleBit,
  _ste: State

  const sceneId = bal.dat?.sceneId || bal.src || 'NONE';
  cpy.activeSceneId = sceneId;
  cpy.stage = 'STAGE_1_INCEPTION_PREFLIGHT';

    bal.dat?.cast || [],
    bal.dat?.requiredVerbs || {}
  );

    ? verifyCanonMerkleLock(bal.dat.canonState, bal.dat.expectedMerkleRoot)
    : { valid: true, calculatedMerkleRoot: 'BYPASS', expectedMerkleRoot: 'BYPASS', diverged: false };

  const isOk = continuity.clean && somatic.valid && canonLock.valid;

  if (!continuity.clean) {
    cpy.stage = 'STAGE_FAILED_DIRTY';
  } else if (!somatic.valid) {
    cpy.stage = 'STAGE_FAILED_SOMATIC';
  } else if (!canonLock.valid) {
    cpy.stage = 'STAGE_FAILED_CANON_DRIFT';
  } else {
    cpy.stage = 'PASSED_PREFLIGHT';
  }

    ok: isOk,
    sceneId,
    continuity,
    somatic,
    canonLock,
    evaluatedAtMs: Date.now(),
    t0Spend: '$0.00',
  };

      lifecycleBit: {
        idx: 'run-inception-preflight',
        val: isOk ? 1 : 0,
        src: isOk ? 'PREFLIGHT_PASSED' : 'PREFLIGHT_BLOCKED',
        dat: { report },
      },
    });
  }
