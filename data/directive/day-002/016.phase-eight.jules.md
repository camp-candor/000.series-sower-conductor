import {
  CameraKinematicsSpec,
  CameraMotionMode,
  JitterOffsetKeyframe,

export function calculateCanvasAnchorOrigin(
  stageX: number,
  stageY: number,
  localAnchorX: number,
  localAnchorY: number,
  scale = 1.0,
  canvasWidth = 1920,
  canvasHeight = 1080
): { originXPercent: number; originYPercent: number } {
  // Sprite local anchor is authored in a 1024x1024 normalized coordinate box
  const normalizedOffsetX = (localAnchorX / 1024) * (canvasWidth * 0.5) * scale;
  const normalizedOffsetY = (localAnchorY / 1024) * (canvasHeight * 0.5) * scale;

  const globalX = stageX + normalizedOffsetX;
  const globalY = stageY + normalizedOffsetY;

  const rawOriginX = (globalX / canvasWidth) * 100;
  const rawOriginY = (globalY / canvasHeight) * 100;

  const originXPercent = Number(Math.max(0, Math.min(100, rawOriginX)).toFixed(2));
  const originYPercent = Number(Math.max(0, Math.min(100, rawOriginY)).toFixed(2));

  return { originXPercent, originYPercent };
}

export function murmurHash3(key: string, seed = 0): number {
  let h1 = seed >>> 0;
  const c1 = 0xcc9e2d51;
  const c2 = 0x1b873593;

  for (let i = 0; i < key.length; i++) {
    let k1 = key.charCodeAt(i);
    k1 = Math.imul(k1, c1);
    k1 = (k1 << 15) | (k1 >>> 17);
    k1 = Math.imul(k1, c2);

    h1 ^= k1;
    h1 = (h1 << 13) | (h1 >>> 19);
    h1 = Math.imul(h1, 5) + 0xe6546b64;
  }

  h1 ^= key.length;
  h1 ^= h1 >>> 16;
  h1 = Math.imul(h1, 0x85ebca6b);
  h1 ^= h1 >>> 13;
  h1 = Math.imul(h1, 0xc2b2ae35);
  h1 ^= h1 >>> 16;

  return h1 >>> 0;
}

export function createMulberry32(seed: number): () => number {
  let state = seed >>> 0;
  return function () {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function synthesizeDeterministicJitterTrajectory(
  sceneId: string,
  shotId: string,
  decayFrames = 14,
  initialIntensityPx = 16.0
): JitterOffsetKeyframe[] {
  const seed = murmurHash3(`${sceneId}:${shotId}:impact_shake`);
  const prng = createMulberry32(seed);
  const trajectory: JitterOffsetKeyframe[] = [];

  for (let t = 0; t < decayFrames; t++) {
    const decay = 1.0 - t / decayFrames;
    const currentIntensity = initialIntensityPx * Math.pow(decay, 1.5);

    const rawDx = (prng() - 0.5) * 2;
    const rawDy = (prng() - 0.5) * 2;

    const deltaX = Number((rawDx * currentIntensity).toFixed(2));
    const deltaY = Number((rawDy * currentIntensity).toFixed(2));

    trajectory.push({ frame: t, deltaX, deltaY });
  }

  return trajectory;
}

export function inspectEjectLockHeader(filePath: string): EjectMarkerStatus {
  if (!fs.existsSync(filePath)) {
    return { isLocked: false, filePath };
  }

  const content = fs.readFileSync(filePath, 'utf8');
  const lines = content.split('\n').slice(0, 10);
  const lockLine = lines.find((l) => l.includes('@showrunner-lock: locked'));

  if (lockLine) {
    return {
      isLocked: true,
      filePath,
      lockAuthor: 'SHOWRUNNER_OVERRIDE',
    };
  }

  return { isLocked: false, filePath };
}

export function synthesizeStaticCompositionTsx(

export const checkEjectLock = (cpy: RemotionModel, bal: RemotionBit, _ste: State) => {
  const cns = '>> [REMOTION]';
  const targetPath = bal.src || bal.dat?.filePath;

  if (!targetPath) {
    const err = 'EJECT_CHECK_REJECTED: Missing filePath';
    process.stderr.write(`${cns} [FAIL] ${err}\n`);
    if (bal.slv) return bal.slv({ remotionBit: { idx: 'check-eject-lock-error', val: 0, src: err } });
    throw new Error(err);
  }

  const status = inspectEjectLockHeader(targetPath);
  cpy.isEjectLocked = status.isLocked;

  if (status.isLocked) {
    process.stdout.write(`${cns} [WARN] Eject lock active for ${path.basename(targetPath)}. Automated mutation blocked.\n`);
  } else {
    process.stdout.write(`${cns} [OK] File unlocked: ${path.basename(targetPath)} available for compilation.\n`);
  }

  if (bal.slv) {
    bal.slv({
      remotionBit: {
        idx: 'check-eject-lock',
        val: status.isLocked ? 1 : 0,
        src: targetPath,
        dat: status,
      },
    });
  }

  return cpy;
};

export const stampStaticComposition = (cpy: RemotionModel, bal: RemotionBit, _ste: State) => {
  const cns = '>> [METAPROGRAM]';
  const sceneId = bal.src || bal.dat?.sceneId || 'ep01_sc01';
  const outputTsxPath = bal.dat?.outputTsxPath || path.resolve(process.cwd(), `drafts/Master_${sceneId}.tsx`);
  const rawShots = bal.dat?.shots || [];
  const kinematicsModes = bal.dat?.kinematicsModes;

  if (rawShots.length === 0) {
    const err = `STAMP_REJECTED: Zero shots provided for scene ${sceneId}`;
    process.stderr.write(`${cns} [FAIL] ${err}\n`);
    if (bal.slv) return bal.slv({ remotionBit: { idx: 'stamp-static-composition-error', val: 0, src: err } });
    throw new Error(err);
  }

  const ejectStatus = inspectEjectLockHeader(outputTsxPath);
  if (ejectStatus.isLocked) {
    const err = `EJECT_BUTTON_LOCKED: Cannot overwrite manually locked composition ${outputTsxPath}`;
    process.stderr.write(`${cns} [FAIL] ${err}\n`);
    if (bal.slv) return bal.slv({ remotionBit: { idx: 'stamp-static-composition-error', val: 0, src: err } });
    throw new Error(err);
  }

  try {
    const timedShots = computeCumulativeShotTimings(rawShots);
    const totalFrames = timedShots.reduce((acc, s) => acc + s.deliveryFrames, 0);

    const tsxCode = synthesizeStaticCompositionTsx(sceneId, totalFrames, timedShots, kinematicsModes);

    const targetDir = path.dirname(outputTsxPath);
    if (!fs.existsSync(targetDir)) {
      fs.mkdirSync(targetDir, { recursive: true });
    }

    const tmpPath = `${outputTsxPath}.tmp`;
    fs.writeFileSync(tmpPath, tsxCode, 'utf8');
    fs.renameSync(tmpPath, outputTsxPath);

    cpy.totalCompositionsStamped += 1;
    cpy.lastStampedPath = outputTsxPath;

    process.stdout.write(
      `${cns} [OK] Stamped static Remotion composition to ${outputTsxPath} (${totalFrames} frames across ${timedShots.length} shots).\n`
    );

    if (bal.slv) {
      bal.slv({
        remotionBit: {
          idx: 'stamp-static-composition',
          val: 1,
          src: outputTsxPath,
          dat: {
            sceneId,
            totalFrames,
            outputTsxPath,
            shotCount: timedShots.length,
          },
        },
      });
    }
  } catch (err: any) {
    process.stderr.write(`${cns} [FAIL] Stamping failed: ${err.message}\n`);
    if (bal.slv) {
      bal.slv({ remotionBit: { idx: 'stamp-static-composition-error', val: 0, src: err.message } });
    } else {
      throw err;
    }
  }

  return cpy;
};
