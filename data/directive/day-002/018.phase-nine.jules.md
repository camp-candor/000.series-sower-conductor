import { formatUnquotedLiteral } from './literalBaking.buzz.js';
import { murmurHash3 } from './kinematics.buzz.js';

export interface DialogueAudioClip {
  clipId: string;
  stemPath: string;
  fromFrame: number;
  durationInFrames: number;
  speakerId: string;
}

export interface FoleyAudioClip {
  cueName: string;
  samplePath: string;
  fromFrame: number;
  durationInFrames: number;
  volumeMultiplier: number;
}

export interface MultiTrackAudioPlan {
  sceneId: string;
  totalDurationFrames: number;
  ambientBedPath: string;
  ambientVolumeLinear: number;
  duckingFloorDb: number;
  dialogueClips: DialogueAudioClip[];
  foleyClips: FoleyAudioClip[];
}

export interface MultiWindowDuckingKeyframe {
  frame: number;
  gainLinear: number;
}

export function computeMultiWindowDuckingCurve(
  totalFrames: number,
  speechWindows: Array<{ startFrame: number; endFrame: number }>,
  duckingFloorDb = -14.0,
  attackFrames = 6,
  releaseFrames = 12
): MultiWindowDuckingKeyframe[] {
  const curve: MultiWindowDuckingKeyframe[] = [];

  for (let f = 0; f < totalFrames; f++) {
    let minGainDb = 0.0;

    for (const win of speechWindows) {
      let windowGainDb = 0.0;

      if (f >= win.startFrame && f <= win.endFrame) {
        // Sustained active vocalization
        windowGainDb = duckingFloorDb;
      } else if (f >= win.startFrame - attackFrames && f < win.startFrame) {
        // Quadratic easing attack (pre-speech fade down)
        const t = (f - (win.startFrame - attackFrames)) / attackFrames;
        windowGainDb = duckingFloorDb * Math.pow(t, 2);
      } else if (f > win.endFrame && f <= win.endFrame + releaseFrames) {
        // Quadratic easing release (post-speech fade up)
        const t = 1.0 - (f - win.endFrame) / releaseFrames;
        windowGainDb = duckingFloorDb * Math.pow(t, 2);
      }

      // Min-reduction across all windows eliminates gain pumping in pauses
      minGainDb = Math.min(minGainDb, windowGainDb);
    }

    const gainLinear = Number(Math.pow(10, minGainDb / 20).toFixed(4));
    curve.push({ frame: f, gainLinear });
  }

  return curve;
}

export function generateMultiTrackAudioJsx(plan: MultiTrackAudioPlan): string {
  const speechWindows = plan.dialogueClips.map((c) => ({
    startFrame: c.fromFrame,
    endFrame: c.fromFrame + c.durationInFrames - 1,
  }));

  const duckingCurve = computeMultiWindowDuckingCurve(
    plan.totalDurationFrames,
    speechWindows,
    plan.duckingFloorDb,
    6,
    12
  );

  const serializedGainCurve = JSON.stringify(duckingCurve.map((k) => k.gainLinear));

  const dialogueSequences = plan.dialogueClips
    .map((c) => {
      const from = formatUnquotedLiteral(c.fromFrame);
      const dur = formatUnquotedLiteral(c.durationInFrames);
      return `      {/* Track 1: Spoken Dialogue (${c.speakerId}) */}
      <Sequence from={${from}} durationInFrames={${dur}}>
        <Audio src={staticFile(${JSON.stringify(c.stemPath)})} volume={1.0} />
      </Sequence>`;
    })
    .join('\n');

  const foleySequences = plan.foleyClips
    .map((f) => {
      const from = formatUnquotedLiteral(f.fromFrame);
      const dur = formatUnquotedLiteral(f.durationInFrames);
      const vol = formatUnquotedLiteral(f.volumeMultiplier);
      const seed = murmurHash3(`${plan.sceneId}:${f.cueName}:${f.fromFrame}`);
      return `      {/* Track 2: Punctual Foley Transient [Seed: ${seed}] */}
      <Sequence from={${from}} durationInFrames={${dur}}>
        <Audio src={staticFile(${JSON.stringify(f.samplePath)})} volume={${vol}} />
      </Sequence>`;
    })
    .join('\n');

  return `
// Multi-Track Acoustic Synchronization Engine (Master Clock Architecture)
export const MultiTrackAudioRig_${plan.sceneId}: React.FC = () => {
  const frame = useCurrentFrame();
  const duckingLut: number[] = ${serializedGainCurve};

  const status = inspectEjectLockHeader(targetPath);
  cpy.isEjectLocked = status.isLocked;

  if (status.isLocked) {
    process.stdout.write(`${cns} [WARN] Eject lock active for ${path.basename(targetPath)}. Automated mutation blocked.\n`);
  } else {
    process.stdout.write(`${cns} [OK] File unlocked: ${path.basename(targetPath)} available for compilation.\n`);
  }

  if (bal.slv) {
    bal.slv({
      remotionBit: {
        idx: 'check-eject-lock',
        val: status.isLocked ? 1 : 0,
        src: targetPath,
        dat: status,
      },
    });
  }

  return cpy;
};

export const stampStaticComposition = (cpy: RemotionModel, bal: RemotionBit, _ste: State) => {
  const cns = '>> [METAPROGRAM]';
  const sceneId = bal.src || bal.dat?.sceneId || 'ep01_sc01';
  const outputTsxPath = bal.dat?.outputTsxPath || path.resolve(process.cwd(), `drafts/Master_${sceneId}.tsx`);
  const rawShots = bal.dat?.shots || [];
  const kinematicsModes = bal.dat?.kinematicsModes;
  const audioPlan = bal.dat?.audioPlan;

  if (rawShots.length === 0) {
    const err = `STAMP_REJECTED: Zero shots provided for scene ${sceneId}`;
    process.stderr.write(`${cns} [FAIL] ${err}\n`);
    if (bal.slv) return bal.slv({ remotionBit: { idx: 'stamp-static-composition-error', val: 0, src: err } });
    throw new Error(err);
  }

  const ejectStatus = inspectEjectLockHeader(outputTsxPath);
  if (ejectStatus.isLocked) {
    const err = `EJECT_BUTTON_LOCKED: Cannot overwrite manually locked composition ${outputTsxPath}`;
    process.stderr.write(`${cns} [FAIL] ${err}\n`);
    if (bal.slv) return bal.slv({ remotionBit: { idx: 'stamp-static-composition-error', val: 0, src: err } });
    throw new Error(err);
  }

  try {
    const timedShots = computeCumulativeShotTimings(rawShots);
    const totalFrames = timedShots.reduce((acc, s) => acc + s.deliveryFrames, 0);

    const tsxCode = synthesizeStaticCompositionTsx(sceneId, totalFrames, timedShots, kinematicsModes, audioPlan);

    const targetDir = path.dirname(outputTsxPath);
    if (!fs.existsSync(targetDir)) {
      fs.mkdirSync(targetDir, { recursive: true });
    }

    const tmpPath = `${outputTsxPath}.tmp`;
    fs.writeFileSync(tmpPath, tsxCode, 'utf8');
    fs.renameSync(tmpPath, outputTsxPath);

    cpy.totalCompositionsStamped += 1;
    cpy.lastStampedPath = outputTsxPath;

    process.stdout.write(
      `${cns} [OK] Stamped static Remotion composition to ${outputTsxPath} (${totalFrames} frames across ${timedShots.length} shots with MultiTrack audio).\n`
    );

    if (bal.slv) {
      bal.slv({
        remotionBit: {
          idx: 'stamp-static-composition',
          val: 1,
          src: outputTsxPath,
          dat: {
            sceneId,
            totalFrames,
            outputTsxPath,
            shotCount: timedShots.length,
          },
        },
      });
    }
  } catch (err: any) {
    process.stderr.write(`${cns} [FAIL] Stamping failed: ${err.message}\n`);
    if (bal.slv) {
      bal.slv({ remotionBit: { idx: 'stamp-static-composition-error', val: 0, src: err.message } });
    } else {
      throw err;
    }
  }

  return cpy;
};

    expect(duckingCurve[0].gainLinear).toBe(1.0);

    expect(duckingCurve[10].gainLinear).toBeCloseTo(0.1995, 2);
