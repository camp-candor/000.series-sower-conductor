    'acute_asphyxiation',
  ]);

  for (const cond of state.active_conditions) {
    if (fatalConditions.has(cond)) {
      throw new Error(`VOCAL_DELIVERY_BREACH: Phonation impossible under active condition [${cond}]`);
    }
  }

  if (state.banned_kinetic_verbs.includes('speak')) {
    throw new Error('VOCAL_DELIVERY_BREACH: Verbal delivery forbidden by active kinetic verb ban');
  }
};

export const mapSomaticToAcousticDsp = (
  state: CharacterSomaticState,
  input: DialogueSomaticInput,
): AcousticDspModifiers => {
  assertPhonationFeasibility(state);

  const fatigueBps = clamp(state.fatigue_bps, 0, 10000);
  const painBps = clamp(state.pain_bps, 0, 10000);

  const basePitch = input.base_pitch_shift_semitones ?? 0.0;
  const cents = input.pitch_shift_cents ?? 0;
  const baseFormant = input.formant_warping_factor ?? 1.0;
  const deltaFormant = input.delta_formant_bps ?? 0;
  const baseBreath = input.base_breathiness ?? 1.0;
  const breathBps = input.breathiness_bps ?? 0;
  const baseFry = input.base_vocal_fry ?? 0.0;
  const fryBps = input.vocal_fry_bps ?? 0;
  const baseSpeed = input.base_speed_scalar ?? 1.0;

  // 1. Pitch Transposition Math: F0_eff = basePitch + (cents/100) + (+3.5 * pain/10000) + (-2.0 * fatigue/10000)
  let pitchShift = basePitch + (cents / 100.0) + (3.5 * (painBps / 10000.0)) - (2.0 * (fatigueBps / 10000.0));
  if (state.active_conditions.includes('crushed_larynx')) {
    pitchShift -= 4.0;
  }
  const effectivePitch = clamp(Number(pitchShift.toFixed(3)), -12.0, 12.0);

  // 2. Formant Warping Factor: W_eff = baseFormant * (1.0 + deltaFormant/10000)
  let formantWarp = baseFormant * (1.0 + (deltaFormant / 10000.0));
  if (state.active_conditions.includes('crushed_larynx')) {
    formantWarp = Math.min(formantWarp, 0.70);
  }
  const effectiveFormant = clamp(Number(formantWarp.toFixed(3)), 0.5, 2.0);

  // 3. Breath Interval Dilation: Pi_eff = baseBreath * (1.0 + 1.40 * (fatigue/10000)^2) * (1.0 + breathBps/10000)
  const fatigueRatio = fatigueBps / 10000.0;
  const breathDilation = baseBreath * (1.0 + 1.40 * Math.pow(fatigueRatio, 2)) * (1.0 + (breathBps / 10000.0));
  const effectiveBreathiness = clamp(Number(breathDilation.toFixed(3)), 1.0, 3.5);

  // 4. Pause Frequency Multiplier & Banned Cadences
  let pauseMultiplier = 1.0;
  const bannedCadences: string[] = [];

  if (fatigueBps >= 8000) {
    pauseMultiplier = 2.8;
    bannedCadences.push('RAPID_STACCATO');
  } else if (fatigueBps > 5000) {
    pauseMultiplier = Number((1.0 + 1.4 * (fatigueBps / 10000.0)).toFixed(3));
  }

  if (state.active_conditions.includes('crushed_larynx')) {
    bannedCadences.push('RISING_INFLECTION_INTERROGATIVE');
  }

  // 5. Tone Pacing Velocity
  const effectivePacing = clamp(Number((baseSpeed * (1.0 - 0.30 * fatigueRatio)).toFixed(3)), 0.1, 3.0);

  // 6. Subharmonic Vocal Fry Density
  const rawFry = baseFry + (0.50 * fatigueRatio) + (0.40 * (painBps / 10000.0)) + (fryBps / 10000.0);
  const effectiveVocalFry = clamp(Number(rawFry.toFixed(3)), 0.0, 1.0);

  return {
    basePitchShiftSemitones: effectivePitch,
    formantWarpingFactor: effectiveFormant,
    breathinessPauseMultiplier: effectiveBreathiness,
    pause_frequency_multiplier: pauseMultiplier,
    vocalFryDensity: effectiveVocalFry,
    tonePacing: effectivePacing,
    target_lufs: -16.0,
    peak_dbtp: -1.0,
    bannedCadences,
  };
};

export const translateSomaticDsp = async (
  cpy: SomaticdspModel,
  bal: SomaticdspBit,
  ste?: State,
): Promise<SomaticdspModel> => {
  const state = bal.dat?.somaticState;
  const input = bal.dat?.somaticInput;

  const emit = async (msg: string) => {
    if (ste && typeof ste.hunt === 'function') {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: msg,
      }).catch(() => {});
    }
  };

  if (!state || !input) {
    const errorMsg = 'TRANSLATE_ERROR: Missing somaticState or somaticInput';
    await emit(`:: [SOMATIC:FAIL] ${errorMsg}`);
    if (bal.slv) {
      bal.slv({
        smcBit: {
          idx: 'translate-somatic-dsp-fail',
          val: -1,
          dat: {
            receipt: {
              ok: false,
              character_id: state?.character_id || 'UNKNOWN',
              dsp_modifiers: {
                basePitchShiftSemitones: 0,
                formantWarpingFactor: 1.0,
                breathinessPauseMultiplier: 1.0,
                pause_frequency_multiplier: 1.0,
                vocalFryDensity: 0.0,
                tonePacing: 1.0,
                target_lufs: -16.0,
                peak_dbtp: -1.0,
                bannedCadences: [],
              },
              error: errorMsg,
            },
          },
        },
      });
    }
    return cpy;
  }

  try {
    const dsp = mapSomaticToAcousticDsp(state, input);

    await emit(
      `>> [SOMATIC:OK] [${state.character_id}]: Pitch=${dsp.basePitchShiftSemitones}st, Formant=${dsp.formantWarpingFactor}, Breath=${dsp.breathinessPauseMultiplier}x, Fry=${dsp.vocalFryDensity}`,
    );

    const receipt: SomaticDspReceipt = {
      ok: true,
      character_id: state.character_id,
      dsp_modifiers: dsp,
    };

    cpy.lastReceipt = receipt;
    cpy.characterProfiles[state.character_id] = state;

    if (bal.slv) {
      bal.slv({
        smcBit: {
          idx: 'translate-somatic-dsp-success',
          val: 0,
          dat: { receipt },
        },
      });
    }
  } catch (err: any) {
    const errorMsg = err instanceof Error ? err.message : String(err);
    await emit(`:: [SOMATIC:FAIL] [${state.character_id}]: ${errorMsg}`);

    const receipt: SomaticDspReceipt = {
      ok: false,
      character_id: state.character_id,
      dsp_modifiers: {
        basePitchShiftSemitones: 0,
        formantWarpingFactor: 1.0,
        breathinessPauseMultiplier: 1.0,
        pause_frequency_multiplier: 1.0,
        vocalFryDensity: 0.0,
        tonePacing: 1.0,
        target_lufs: -16.0,
        peak_dbtp: -1.0,
        bannedCadences: [],
      },
      error: errorMsg,
    };

    cpy.lastReceipt = receipt;

    if (bal.slv) {
      bal.slv({
        smcBit: {
          idx: 'translate-somatic-dsp-fail',
          val: -1,
          dat: { receipt },
        },
      });
    }
  }

  return cpy;
};
